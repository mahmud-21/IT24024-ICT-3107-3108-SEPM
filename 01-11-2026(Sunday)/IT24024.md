# The code of Prime Number Dao
```java
package com.example.dao;

import com.example.util.DBConnection;

import java.sql.Connection;
import java.sql.PreparedStatement;
import java.sql.SQLException;

public class PrimeNumberDAO {

    public void savePrimeNumber(int number) throws SQLException {

        String sql = "INSERT INTO prime_numbers (number) VALUES (?)";

        try (Connection connection = DBConnection.getConnection();
             PreparedStatement statement =
                     connection.prepareStatement(sql)) {

            statement.setInt(1, number);

            int rowsInserted = statement.executeUpdate();

            if (rowsInserted != 1) {
                throw new SQLException("Failed to save the prime number.");
            }
        }
    }
}
```

# The code of PrimeServlet
```java
package com.example.servlet;

import com.example.dao.PrimeNumberDAO;
import com.example.util.PrimeUtil;

import jakarta.servlet.ServletException;
import jakarta.servlet.annotation.WebServlet;
import jakarta.servlet.http.HttpServlet;
import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;

import java.io.IOException;
import java.io.PrintWriter;
import java.sql.SQLException;

@WebServlet("/prime")
public class PrimeServlet extends HttpServlet {

    @Override
    protected void doPost(
            HttpServletRequest request,
            HttpServletResponse response)
            throws ServletException, IOException {

        response.setContentType("text/html;charset=UTF-8");

        String input = request.getParameter("number");
        int n;

        try {
            n = Integer.parseInt(input == null ? "" : input.trim());

            if (n < 1 || n > 100000) {
                throw new NumberFormatException();
            }
        } catch (NumberFormatException e) {
            response.sendError(
                    HttpServletResponse.SC_BAD_REQUEST,
                    "Enter a whole number between 1 and 100000."
            );
            return;
        }

        int nthPrime = PrimeUtil.findNthPrime(n);

        try {
            PrimeNumberDAO dao = new PrimeNumberDAO();
            dao.savePrimeNumber(nthPrime);
        } catch (SQLException e) {
            throw new ServletException(
                    "Could not save the result to MySQL.", e
            );
        }

        PrintWriter out = response.getWriter();

        out.println("<!DOCTYPE html>");
        out.println("<html lang='en'>");
        out.println("<head>");
        out.println("<meta charset='UTF-8'>");
        out.println("<title>Nth Prime Result</title>");
        out.println("</head>");
        out.println("<body>");
        out.println("<h1>Nth Prime Number Finder</h1>");
        out.println("<p>You entered: <strong>" + n + "</strong></p>");
        out.println("<p>Prime number at position " + n
                + ": <strong>" + nthPrime + "</strong></p>");
        out.println("<p>Saved to MySQL successfully.</p>");
        out.println("<a href='" + request.getContextPath()
                + "/index.jsp'>Go Back</a>");
        out.println("</body>");
        out.println("</html>");
    }
}
```
